#include <iostream>
#include <vector>
#include <chrono>
#include <cmath>
#include <omp.h>


// 1 задача
#define N 1000000

int main() {
    setlocale(LC_ALL, "RUS");
    double* a, * b;
    double dot_product = 0.0;
    double start_time, end_time;
    int num_threads;

    // Выделение памяти
    a = (double*)malloc(N * sizeof(double));
    b = (double*)malloc(N * sizeof(double));

    // Инициализация векторов
    for (int i = 0; i < N; i++) {
        a[i] = 1.0;
        b[i] = 2.0;
    }

    start_time = omp_get_wtime();

    // Получаем количество потоков
#pragma omp parallel
    {
#pragma omp single
        num_threads = omp_get_num_threads();
    }

    // Массив для частичных сумм каждого потока
    double* partial_sums = (double*)calloc(num_threads, sizeof(double));

#pragma omp parallel
    {
        int thread_id = omp_get_thread_num();

#pragma omp for
        for (int i = 0; i < N; i++) {
            partial_sums[thread_id] += a[i] * b[i];
        }
    }

    // Суммирование результатов всех потоков
    for (int i = 0; i < num_threads; i++) {
        dot_product += partial_sums[i];
    }

    end_time = omp_get_wtime();

    printf("Скалярное произведение: %f\n", dot_product);
    printf("Время выполнения: %f секунд\n", end_time - start_time);
    printf("Количество потоков: %d\n", num_threads);

    // Очистка памяти
    free(a);
    free(b);
    free(partial_sums);

    return 0;
}
